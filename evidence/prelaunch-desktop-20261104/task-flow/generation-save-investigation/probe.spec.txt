test('investigation: delayed preparation replies serialize real clicks and preserve accepted selection and text',async({page})=>{
 await page.setViewportSize({width:1440,height:960});
 const state=await setup(page,true);
 await page.getByRole('button',{name:'调整生成范围',exact:true}).click();
 const held:{body:any;release:()=>void}[]=[];
 await page.route('**/api/v1/projects/*/file-generation/preparation',async route=>{
  const body=fixtureRequest(route.request());
  expect(body.expected_revision).toBe(state.data.revision);
  const snapshot=structuredClone(body.preparation);
  state.data.revision++;state.data.preparation=snapshot;
  const response={revision:state.data.revision,preparation:snapshot};
  await new Promise<void>(release=>held.push({body:structuredClone(body),release}));
  return route.fulfill({json:fixtureBody(route,response)});
 });
 await page.getByRole('button',{name:'取消全选',exact:true}).click();
 await expect.poll(()=>held.length).toBe(1);
 expect(held[0].body.preparation.item_ids).toEqual([]);
 await expect(page.locator('#fg-controls')).toHaveJSProperty('inert',true);
 // The busy paint happens before the mutation and temporarily shows the old selection.
 await expect(page.locator('[data-fg-item="entry-1"]')).toBeChecked();
 const blocked=await page.locator('[data-fg-item="entry-1"]').boundingBox();
 await page.mouse.click(blocked!.x+blocked!.width/2,blocked!.y+blocked!.height/2);
 expect(held).toHaveLength(1);
 held[0].release();
 await expect(page.locator('#fg-controls')).toHaveJSProperty('inert',false);
 await expect(page.locator('.fg-footer')).toContainText('本批 0 份文件');
 await expect(page.locator('[data-fg-item="entry-1"]')).not.toBeChecked();
 // Use click, not check(): a real click is accepted while check() may inspect the busy replacement DOM.
 await page.locator('[data-fg-item="entry-1"]').click();
 await expect.poll(()=>held.length).toBe(2);
 expect(held[1].body.preparation.item_ids).toEqual(['entry-1']);
 await expect(page.locator('#fg-controls')).toHaveJSProperty('inert',true);
 await expect(page.locator('[data-fg-item="entry-1"]')).not.toBeChecked();
 await expect(page.locator('.fg-footer')).toContainText('本批 0 份文件');
 await mkdir('../../evidence/prelaunch-desktop-20261104/task-flow/generation-save-investigation',{recursive:true});
 await page.screenshot({path:'../../evidence/prelaunch-desktop-20261104/task-flow/generation-save-investigation/busy-transient-1440x960.png'});
 const note=await page.locator('#fg-requirements').boundingBox();
 await page.mouse.click(note!.x+20,note!.y+20);await page.keyboard.insertText('busy期间输入不会进入表单');
 expect(held).toHaveLength(2);expect(await page.locator('#fg-requirements').inputValue()).toBe('');
 held[1].release();
 await expect(page.locator('#fg-controls')).toHaveJSProperty('inert',false);
 await expect(page.locator('[data-fg-item="entry-1"]')).toBeChecked();
 await expect(page.locator('.fg-footer')).toContainText('本批 1 份文件');
 const requirements='请保留待核实事项；金额单位为万元。';
 await page.locator('#fg-requirements').fill(requirements);
 await page.getByRole('button',{name:'保存准备清单',exact:true}).click();
 await expect.poll(()=>held.length).toBe(3);
 expect(held[2].body.preparation.item_ids).toEqual(['entry-1']);expect(held[2].body.preparation.requirements).toBe(requirements);
 held[2].release();
 await expect(page.locator('#fg-controls')).toHaveJSProperty('inert',false);
 await expect(page.locator('#fg-requirements')).toHaveValue(requirements);
 await expect(page.locator('[data-fg-item="entry-1"]')).toBeChecked();
 await page.screenshot({path:'../../evidence/prelaunch-desktop-20261104/task-flow/generation-save-investigation/accepted-saved-1440x960.png'});
 expect(state.quotes).toBe(0);expect(state.submissions).toHaveLength(0);
 await page.reload();
 await expect(page.locator('#fg-requirements')).toHaveValue(requirements);
 await expect(page.locator('.fg-footer')).toContainText('本批 1 份文件');
 expect(held).toHaveLength(3);
 expect(state.transport.errors).toEqual([]);expect(state.transport.unexpected).toEqual([]);
});
